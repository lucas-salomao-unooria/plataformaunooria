\section{Como uma missão é feita}

Todas as missões seguem o mesmo desenho, e é ele que dá identidade ao
produto:

\begin{enumerate}
  \item \textbf{O disfarce.} O estudante entra num aplicativo que conhece
    (rede social de fotos, grupo de mensagens, assistente de IA) ou num jogo,
    e age como agiria no próprio celular.
  \item \textbf{A decisão.} Em algum momento ele escolhe --- o que repassar,
    em quem confiar, o que salvar --- sem saber que aquilo será olhado depois.
  \item \textbf{A revelação.} A missão deixa de fingir e mostra o que estava
    acontecendo por trás. Em algumas, a revelação é da própria tela; em
    outras, acontece no debate conduzido pelo professor.
  \item \textbf{O debate.} Os aparelhos congelam e a turma conversa. É aqui
    que a aprendizagem acontece, e por isso quase toda missão tem um momento
    que só o professor destrava.
  \item \textbf{O percurso.} O estudante registra o que levou da missão (o
    \enquote{Meu percurso}), sem nota.
\end{enumerate}

\feature{F06.1}{O Que a Foto Conta}{\parcial}

\textbf{1º ano EF, Unidade 1} \textperiodcentered{} imita uma rede social de
fotos (o \enquote{mural da turma}) \textperiodcentered{} uma aula
\textperiodcentered{} IA só no preparo do professor.

\textbf{Proposta.} Letramento visual para quem está se alfabetizando: toda
foto é uma escolha de enquadramento e legenda, e o que fica de fora também
conta a história. É 100\% formativa: não há foto errada nem nota.

\begin{historia}{H06.1.1}{Publicar uma foto no mural da turma}{\parcial}
\hu{Estudante do 1º ano}{escolher uma foto e uma legenda e publicar no
mural}{descobrir que cada foto mostra só uma parte da história}
\aceite
\begin{itemize}
  \item Nina dá as boas-vindas em áudio e a criança escolhe a sua marca
    (Coruja Azul, Estrela Amarela ou Sol Laranja).
  \item Explora um mural com 3 fotos e curte com toque duplo; parada por 30
    segundos, uma seta indica o caminho.
  \item Recebe duas fotos do mesmo pátio (plano geral com peças espalhadas
    \emph{versus} detalhe do brinquedo montado) e duas legendas; escolhe uma
    de cada e toca em COMPARTILHAR.
  \item Depois de publicar, os colegas simulados Leo e Sofia comentam o que a
    foto mostrou e o que escondeu.
  \item Todo texto em CAIXA ALTA, com no máximo 8 palavras, pictograma e
    áudio.
\end{itemize}
\pendencia{as fotos ainda são desenhos provisórios e as locuções
profissionais não foram gravadas (a tela mostra texto e ícone). O botão de
repetir o áudio já é maior, do tamanho do dedo de uma criança de 6 anos.}
\end{historia}

\begin{historia}{H06.1.2}{Refletir sobre o que a foto mostra}{\pronto}
\hu{Estudante do 1º ano}{responder o que a foto mostra depois da roda de
conversa}{guardar a ideia de que olhar é escolher}
\aceite
\begin{itemize}
  \item Todos os tablets congelam em \enquote{HORA DA RODA DE CONVERSA!}, uma
    tela sem nenhum botão.
  \item Depois, a reflexão \enquote{O QUE A FOTO MOSTRA?}: a resposta Lupa
    conclui com confete e a insígnia \enquote{Olhar de Detetive}; a resposta
    Globo recebe um lembrete gentil e a criança tenta de novo, sem limite.
\end{itemize}
\end{historia}

\begin{historia}{H06.1.3}{Adaptar a missão ao tema da semana}{\pronto}
\hu{Professor}{gerar legendas e comentários sobre o tema que a turma está
estudando}{que a missão converse com a minha aula}
\aceite
\begin{itemize}
  \item Temas: Pátio do Recreio, Horta, Sala de Aula, Parquinho.
  \item Gero o pacote com IA, confiro e aprovo; nada chega à turma sem
    aprovação.
  \item Se eu não fizer nada, roda o pacote pronto \enquote{O Mistério das
    Peças no Pátio}.
  \item A IA nunca roda no tablet da criança e nunca a avalia.
\end{itemize}
\end{historia}

\begin{historia}{H06.1.4}{Conduzir a roda e ver quem hesitou}{\pronto}
\hu{Professor}{conduzir a roda de conversa e saber quem teve dificuldade de
leitura}{ajudar quem precisa, sem expor ninguém}
\aceite
\begin{itemize}
  \item Projeto as duas fotos lado a lado e conduzo a roda; depois toco em
    \enquote{LIBERAR REFLEXÃO FINAL}.
  \item Vejo a lista de \enquote{hesitação de leitura} (mais de 60 segundos na
    decisão), com nome, só para mim.
  \item \enquote{Zerar a turma} arquiva a tentativa, em vez de apagar.
\end{itemize}
\end{historia}

\feature{F06.2}{UnooTube --- Em quais fontes você confiaria?}{\arquivada}

\begin{nota}
\textbf{Arquivada em 25 de setembro de 2026.} O Jornal da Escola passou a ocupar
a Unidade 1 do 4º ano (\id{H06.10.4}), e o UnooTube saiu da trilha sem ser
apagado: não abre para nenhuma turma, os dados que as turmas produziram ficaram
guardados, e a equipe da Unooria vê a missão na lista de arquivadas
(\id{H10.6.1}). O que segue descreve a missão como ela era.
\end{nota}

\textbf{4º ano EF, Unidade 1} \textperiodcentered{} imita um aplicativo de
vídeos curtos, em tela de computador \textperiodcentered{} uma aula
\textperiodcentered{} sem IA.

\textbf{Proposta.} Distinguir fonte confiável de sensacionalismo e perceber
que ter muitos inscritos não significa contar a verdade.

\begin{historia}{H06.2.1}{Salvar as três fontes em que eu confiaria}{\arquivada}
\hu{Estudante do 4º ano}{assistir aos vídeos e salvar os três em que eu
confiaria}{fazer uma escolha de verdade antes da revelação}
\aceite
\begin{itemize}
  \item Um feed de 21 vídeos curtos de História, em ordem aleatória; posso
    ligar o som, abrir a descrição, curtir e salvar.
  \item A pesquisa do aplicativo está bloqueada de propósito.
  \item Cabem 3 vídeos na bandeja; ao tentar um quarto, o aviso \enquote{sua
    playlist tem três lugares} oferece trocar.
  \item No terceiro salvo, habilita \enquote{Concluir missão} --- quem decide
    quando fechar é o estudante.
\end{itemize}
\end{historia}

\begin{historia}{H06.2.2}{Ver o ranking virar}{\arquivada}
\hu{Estudante do 4º ano}{comparar as minhas escolhas com as da turma}{perceber
que popularidade não é credibilidade}
\aceite
\begin{itemize}
  \item O Ranking da turma ordena por inscritos, com colegas fictícios: quem
    escolheu vídeos falsos aparece no topo.
  \item \enquote{Adicionar credibilidade} revela a coluna de fontes confiáveis,
    e o ranking vira.
  \item Depois disso os salvos congelam; reiniciar descongela.
  \item Os links de fonte aparecem como texto: o estudante não sai da
    plataforma.
\end{itemize}
\end{historia}

\begin{historia}{H06.2.3}{Ler a turma}{\arquivada}
\hu{Professor}{ver quantas fontes confiáveis a turma escolheu}{saber por onde
começar o debate}
\aceite
\begin{itemize}
  \item Não há nada a configurar além de liberar.
  \item Resultados: \enquote{Estudantes na turma}, \enquote{Média de fontes
    confiáveis} e a distribuição de 0 a 3 fontes confiáveis.
  \item O guia traz a tabela dos 21 vídeos, com quais são confiáveis.
\end{itemize}
\pendencia{os vídeos precisam estar publicados no armazenamento de cada rede;
sem isso, o feed aparece sem vídeo.}
\end{historia}

\feature{F06.11}{Quem Mandou Isso? --- versão nova}{\parcial}

\textbf{5º ano EF, Unidade 1} \textperiodcentered{} imita um mensageiro, em
tema claro, com a identidade da Unooria \textperiodcentered{} uma aula de 50 minutos
\textperiodcentered{} IA só no preparo do professor \textperiodcentered{}
aceita equipes.

\textbf{Proposta.} Perceber que o jeito de uma mensagem --- séria ou alarmada
--- não diz se ela é verdadeira, e montar a própria regra para conferir antes
de repassar. É a primeira missão que registra o que cada estudante demonstrou
das habilidades da BNCC (\id{E14}).

\begin{historia}{H06.11.1}{Preparar o pacote de mensagens da turma}{\parcial}
\hu{Professor}{preparar as oito mensagens do grupo sobre um assunto da minha
escola}{que a turma leia uma confusão que parece a dela, e não uma qualquer}
\aceite
\begin{itemize}
  \item Em \enquote{A aula e os resultados}, escolho o componente curricular,
    escrevo o tema --- ele completa a fala da Nina, \enquote{Rolou uma confusão
    sobre\ldots}, com até 60 caracteres e 8 palavras ---, listo de 3 a 6 fatos
    da aula, um por linha --- a tela mostra a lista dos fatos que reconheceu, e
    o erro pede o Enter quando estão todos numa linha só ---, e, se quiser, o
    contexto da escola.
  \item \enquote{Gerar Atividade com IA} escreve as oito mensagens; sem saldo
    ou com a IA fora do ar, a tela avisa e já traz o pacote pronto da Unooria,
    \enquote{O Mistério do Passeio ao Museu}. \enquote{Usar Pacote Curado
    Padrão} o traz também, sem gerar nada.
  \item Vejo as oito em quatro grupos --- verdadeira ou falsa, emocional ou
    neutra ---, com os selos que só eu vejo, e posso reescrever qualquer texto;
    a resposta de cada mensagem não se edita.
  \item Toda turma recebe duas mensagens de cada grupo, quatro textos, duas
    imagens e dois links --- e a imagem e o link têm uma verdadeira e uma falsa
    cada, para o formato não entregar a resposta.
  \item Abaixo das mensagens, reviso as três falas do podcast de
    recapitulação que contam este grupo (\id{H06.11.6}); \enquote{Criar
    atividade} espera por elas e as aprova junto.
  \item Nada chega à turma sem o meu clique em \enquote{Criar atividade}, no
    fim da criação; sem ele, a criança abre a missão e vê a lista de conversas
    vazia, com um aviso de espera.
  \item Aprovar um pacote diferente pede confirmação: a turma recomeça com as
    mensagens novas, e o que ela fez com o anterior fica guardado.
  \item A tela diz se a missão já está liberada na sala da turma, e leva à
    ficha para montar as equipes.
\end{itemize}
\pendencia{textos escritos pela equipe --- a prévia do link do pacote pronto,
os fatos dele e a lista de componentes curriculares --- aguardam a revisão do
PO; o equilíbrio de imagem e link no pacote pronto espera a decisão \id{D14}.}
\end{historia}

\begin{historia}{H06.11.2}{Escolher exatamente três mensagens para o Mural}{\pronto}
\hu{Estudante do 5º ano}{ler o grupo da turma e mandar ao Mural da Turma as
três mensagens que explicam o que aconteceu}{fazer o que eu faria no meu
celular, sem ninguém me dizer em que desconfiar}
\aceite
\begin{itemize}
  \item A missão abre na lista de conversas; a Nina pede ajuda sobre o tema da
    turma e manda o convite do grupo \enquote{Turma 5ºB}.
  \item A Nina aparece com foto, na lista e no chat.
  \item No grupo, as oito mensagens chegam uma a uma, com \enquote{digitando\ldots}
    antes de cada uma, e cada colega tem o nome numa cor; as duas imagens têm
    desenhos diferentes (comunicado, cartaz, print ou bilhete). Ao voltar do
    Mural, o grupo já está inteiro.
  \item Nenhuma mensagem chega marcada; só
    administradores escrevem, e a única ação é a seta de encaminhar, que abre
    \enquote{Encaminhar para\ldots} com um destino só: o Mural da Turma.
  \item Encaminho exatamente três. A quarta seta diz \enquote{Limite de 3
    mensagens atingido. Desmarque uma mensagem para escolher outra.}, e a faixa
    do topo, com menos de três, diz o que falta.
  \item No Mural as três aparecem como \enquote{Encaminhada}, sem o nome de
    quem mandou. Posso voltar ao grupo e trocar até tocar em \enquote{Confirmar
    envio}; depois, não muda mais.
  \item Parado 90 segundos sem encaminhar, a Nina lembra como encaminhar --- o
    gesto, nunca a escolha.
  \item Se eu recarregar a página, volto onde estava, com as escolhas
    guardadas.
\end{itemize}
\end{historia}

\begin{historia}{H06.11.3}{Dizer o que pesou e descobrir os dois selos}{\pronto}
\hu{Estudante do 5º ano}{dizer por que escolhi aquelas mensagens e depois ver o
que cada uma era}{perceber que mensagem séria pode ser falsa, e alarmada pode
ser verdadeira}
\aceite
\begin{itemize}
  \item Confirmado o Mural, respondo à enquete \enquote{O que mais te fez
    escolher essas mensagens?}, com quatro opções, e espero a turma.
  \item Quando o professor libera a revelação, cada uma das minhas três ganha
    dois selos --- verdadeira ou falsa, emocional ou neutra ---, com o motivo
    que eu votei, um lembrete do bimestre e as duas barras da turma, sem nome
    de ninguém. Chega sozinho, sem recarregar a página.
  \item Antes do meu voto e da revelação, a resposta de nenhuma mensagem chega
    ao meu aparelho.
  \item Quem ainda estava escolhendo quando o professor liberou continua,
    confirma, vota e vê os próprios selos depois.
  \item A síntese chega ao aparelho quando o professor a apresenta, com o aviso
    de participar da conversa na sala.
\end{itemize}
\end{historia}

\begin{historia}{H06.11.4}{Refletir com as perguntas da Nina}{\parcial}
\hu{Estudante do 5º ano}{responder às perguntas da Nina sobre o que
aconteceu}{pensar no que eu faria antes de repassar uma mensagem}
\aceite
\begin{itemize}
  \item Com a reflexão liberada, o chat da Nina abre direto na primeira
    pergunta, sem repetir o pedido do começo; não há regra para escrever.
  \item Respondo a duas perguntas rápidas com a mesma enquete do Mural: se
    concordo com a colega que confia na mensagem \enquote{porque é séria}, e
    qual de três regras funciona de verdade. Depois de cada uma, vejo só
    \enquote{Resposta enviada} --- nunca se acertei.
  \item A Nina se despede, e a tela da escola mostra \enquote{Missão
    Concluída}.
\end{itemize}
\pendencia{a despedida da Nina é texto da equipe e aguarda a revisão do PO.}
\end{historia}

\begin{historia}{H06.11.5}{Conduzir a aula e ler a turma}{\parcial}
\hu{Professor}{conduzir a aula em três momentos e ver o que a turma
escolheu}{abrir a roda de conversa com o retrato da sala, sem expor ninguém}
\aceite
\begin{itemize}
  \item Três botões, na ordem --- \enquote{Liberar Revelação},
    \enquote{Apresentar Síntese} e \enquote{Liberar Reflexão} ---, cada um com
    confirmação e sempre para a frente; a síntese pode ser pulada.
  \item O painel se atualiza sozinho: quantos entraram no grupo, confirmaram o
    Mural, votaram e concluíram.
  \item Depois da revelação, o Termômetro da Turma: quantos repassaram uma
    falsa de formato sério, quantas verdadeiras exageradas ficaram de fora e o
    cruzamento com o motivo votado. \enquote{Projetar o Termômetro} abre uma
    tela cheia sem nome nenhum; com ninguém que confirmou, \enquote{sem dados},
    nunca 0\%.
  \item O roteiro da roda de conversa, com três perguntas.
  \item A folha da turma: onde cada um está, as três mensagens com o grupo de
    cada uma e o motivo. Sem coluna de acerto.
  \item O bloco \enquote{Habilidades da BNCC} mostra a EF05CO08 pela escolha do
    Mural e a EF05CO03 e a EF05CO04 pelas duas perguntas da Nina, e é ali que
    confirmo e ajusto (\id{E14}).
  \item \enquote{Zerar Turma para Nova Aplicação} recomeça a turma numa rodada
    nova, com o mesmo pacote, depois de uma confirmação deliberada; as
    habilidades do pacote são apagadas, e a rodada nova avalia a turma do zero
    (\id{H14.2.4}).
  \item Se a turma divide computadores, monto as equipes na criação da
    atividade, e o painel conta por computador (\id{E13}).
  \item O guia da aula traz o pacote pronto com as respostas, o roteiro e as
    perguntas da Nina com o que cada alternativa vale.
\end{itemize}
\pendencia{o texto do botão da síntese é da equipe e aguarda o PO.}
\end{historia}

\begin{historia}{H06.11.6}{O podcast de recapitulação}{\parcial}
\hu{Professor do 5º ano}{um episódio curto em que dois apresentadores
recapitulam a missão com a turma, contando o grupo que ela leu}{fechar a
conversa depois da aula, ou retomá-la na aula seguinte, com a turma ouvindo a
mesma coisa}
\aceite
\begin{itemize}
  \item O episódio tem cerca de 3 minutos e 40 segundos, com Lucas e Camila, e
    segue o roteiro aprovado para a missão, com o grupo no Unoochat. É do
    professor --- nenhum estudante o recebe ---, e ele decide se e quando toca
    para a turma.
  \item Só três falas mudam com o tema, porque contam o que a turma viu: o que
    o grupo trazia, a mensagem calma que não era verdadeira e a mensagem
    desesperada que era. O resto do roteiro é igual para todas as turmas.
  \item \textbf{Reviso as três falas ao criar a atividade}, abaixo das oito
    mensagens (\id{H06.11.1}): quando as mensagens chegam à tela, as falas
    chegam junto, escritas pela IA a partir delas, e eu posso editar cada uma
    ou pedir que sejam escritas de novo. Com o pacote pronto da Unooria, sem
    mudança, as falas são as do roteiro aprovado, sem IA. \enquote{Criar
    atividade} espera as falas, e elas são aprovadas junto com as mensagens.
  \item Se o limite do dia acabar ou a IA estiver fora do ar, a tela traz as
    falas do pacote pronto e avisa que elas contam outro grupo, para eu
    reescrever.
  \item Se eu mudar as mensagens depois de as falas serem escritas, a tela
    avisa que elas podem não contar mais o que a turma vai ver.
  \item Na página da atividade, em \enquote{A aula e os resultados}, fica o
    podcast: a transcrição inteira, bloco a bloco, mesmo sem áudio, e
    \enquote{Gerar o áudio do podcast}, que cria a voz com IA um trecho por
    vez e conta no meu limite diário. O que é igual para todas as turmas é
    gerado uma vez só.
  \item Uma atividade criada com IA antes de o podcast existir não tem falas
    aprovadas: a página dela diz que é preciso criar a atividade de novo para
    ter o podcast.
  \item O guia da aula traz o passo \enquote{Gere o áudio do podcast, se for
    usá-lo}.
\end{itemize}
\pendencia{a voz gerada precisa da revisão do conteúdo antes da primeira rede;
se não passar, o episódio é gravado em estúdio, com o mesmo roteiro.}
\end{historia}

\textperiodcentered{} IA só no preparo do professor, com mostra pronta da
Unooria \textperiodcentered{} aceita equipes.

\textbf{Proposta.} Viver por dentro o que um feed faz com quem decide por ele.
O estudante é jurado de uma mostra científica estadual: julga os estandes pelo
feed, vota com certeza no melhor e descobre, na \textbf{Galeria Oculta}, que
cada colega julgou uma feira diferente --- montada pelas próprias curtidas.
Depois desmonta a engrenagem: a curtida vira dado, o dado alimenta o algoritmo
e o algoritmo fecha a bolha. É a quarta missão que registra o que cada
estudante demonstrou das habilidades da BNCC (\id{E14}): a EF06CO10, o foco ---
olhar com senso crítico a plataforma que escolhe o que eu vejo ---, a EF06CO01
e a EF06CO03. \textbf{Desde 10 de outubro de 2026 a missão segue o PRD v2 do
PO}: volta a mostra pronta da Unooria (dezesseis projetos), a cédula fica com
quatro elementos e a revisão com dois botões, e saem a justificativa por
cartões, o complementar e a rodada de desvio que tinham entrado em 7 de
outubro. O PO aprovou nove ajustes ao PRD (por exemplo: não há
\enquote{Pausar Sessão} nem \enquote{Exportar}, que a plataforma não tem).
\textbf{Desde 14 de outubro de 2026 a mostra tem quarenta projetos}, dez por
área, aprovados pelo PO: o feed não repete post nenhum (antes, a área preferida
voltava com os mesmos projetos em outros formatos), e a Galeria Oculta mostra
O QUE sumiu --- quantos estandes de cada área chegaram ao feed e o feed que o
estudante teria visto sem as próprias curtidas ---, porque a quantidade
escondida passou a ser a mesma para todos.

\begin{historia}{H06.14.1}{Preparar a mostra da turma}{\parcial}
\hu{Professor}{usar a mostra pronta da Unooria, ou gerar uma sobre o que a
turma está estudando e revisar os estandes antes da aula}{que a turma julgue
projetos que conversam com a minha aula}
\aceite
\begin{itemize}
  \item Crio a atividade pela \enquote{Nova atividade} da ficha (\id{F06.13}).
    O caminho curto vem primeiro: \enquote{Usar Banco Padrão Oficial} cria a
    atividade com os quarenta projetos prontos da Unooria, dez em cada área
    (Tecnologia, Meio Ambiente, Cultura e Saúde), já aprovados, com as imagens
    que a Unooria gerou para eles (\id{H06.14.9}); o estande que ainda não tem
    imagem aparece com a arte desenhada da missão.
  \item Ou, em \enquote{Configuração Pedagógica da Mostra}, escolho o
    componente curricular, escrevo o tema da feira, de três a seis assuntos da
    semana e, se quiser, um contexto da escola, e \enquote{Gerar Mostra
    Personalizada com IA} monta quatro áreas e quarenta estandes, dez por
    área, em cerca de um minuto. Os estandes aparecem numa aba por área, e a
    aba avisa quantos daquela área precisam de ajuste. Leio, edito e posso
    \enquote{Regenerar} um estande; a tela recusa o que foge das regras e diz o
    que corrigir.
  \item Se o limite do dia de IA acabou ou a IA está fora do ar, a tela avisa
    e oferece a mostra pronta (\enquote{Usar Acervo Padrão da Mostra}): a aula
    não fica sem mostra.
  \item A aprovação da mostra gerada tem \textbf{duas fases}: os textos (com
    as falas do podcast, \id{H06.14.7}) e depois as imagens, na grade da
    página da atividade, também com uma aba por área. As quarenta imagens
    saem quatro por vez e levam vários minutos: o guia recomenda preparar
    antes do dia da aula. Sem as imagens aprovadas, a atividade não vai ao ar;
    aprovadas, elas não mudam mais.
  \item A página da atividade mostra os estandes também separados por área,
    numa aba para cada uma.
\end{itemize}
\pendencia{os vinte e quatro projetos novos da mostra pronta foram escritos pela
equipe no molde dos dezesseis do PO e ainda esperam a leitura dele; e o teste
que percorre a missão como professora e dois estudantes precisa ser atualizado
para os quarenta estandes.}
\end{historia}

    depois de inspecionar ao menos um escondido) ou \enquote{Confirmar Voto

\end{itemize}

\feature{F06.12}{A Seleção}{\parcial}

\textbf{7º ano EF, Unidade 1} \textperiodcentered{} imita uma rede social de
fotos, em moldura de celular \textperiodcentered{} uma aula
\textperiodcentered{} IA no preparo do professor, no podcast e na leitura da
legenda \textperiodcentered{} aceita equipes.

\textbf{Proposta.} Perceber que publicar é recortar. Três veículos cobrem o
mesmo acontecimento, cada um pela sua lente --- a oficial, a da comunidade e a
dos números ---, e quem escolhe três posts conta uma história tão verdadeira
quanto a do colega que escolheu outros três. Desde outubro de 2026 a rede
imitada é o \textbf{Unoovista}, a mesma do 8º ano, com a identidade visual da
Unooria; até ali era o Instanooria. O que o estudante faz não mudou. É a segunda missão que registra o
que cada estudante demonstrou das habilidades da BNCC (\id{E14}), e a única em
que a IA dá um nível, como exceção declarada (\id{H09.3.1}).

\begin{historia}{H06.12.1}{Preparar a cobertura da turma}{\pronto}
\hu{Professor}{escolher o acontecimento que os três veículos cobrem e revisar
os 12 posts antes da aula}{que a turma leia uma cobertura que pareça a da
cidade dela}
\aceite
\begin{itemize}
  \item A cobertura é a atividade, e quem a cria sou eu, pela \enquote{Nova
    atividade} da ficha (\id{F06.13}). A cobertura pronta da Unooria, sobre a
    reforma da praça central, é um ponto de partida que eu leio e edito: sem
    atividade em andamento, a turma vê a missão fechada.
  \item Em \enquote{Preparação da Missão: A Seleção}, escrevo o
    \enquote{Evento central coberto pelos veículos} e, se quiser, \enquote{Um
    detalhe da sua cidade ou escola (opcional)}. \enquote{Gerar Lote com IA}
    escreve os 12 posts, quatro de cada veículo, e as três falas do podcast que
    dependem do tema (\id{H06.12.6}); sem saldo ou com a IA fora do ar, a tela
    avisa e carrega a cobertura pronta.
  \item Leio e edito os 12 posts, cada um num card com as fichas que só eu
    vejo (a fonte ouvida e o ângulo). Cada legenda tem de 20 a 45 palavras, e a
    tela aponta a que sair do limite. Os dois comentários de leitor de cada post
    vêm com quem os escreveu, um apelido inventado de rede social.
  \item No lugar de cada imagem, o card avisa \enquote{A imagem será gerada
    após a aprovação dos textos}: aprovar os textos é a primeira fase, e só
    depois dela as imagens são criadas (\id{H06.12.7}) --- imagem de um
    rascunho gastaria o meu limite à toa.
  \item \enquote{Aprovar textos e gerar as imagens} guarda a cobertura. Ela só
    chega à turma depois de eu aprovar também as imagens; até lá, aparece na
    lista como \enquote{Em preparo}. A cobertura que estava no ar fica
    guardada.
\end{itemize}
\end{historia}

\end{historia}

\begin{historia}{H06.12.3}{Ver a minha seleção ao lado da de um colega}{\pronto}
\hu{Estudante do 7º ano}{comparar os meus Stories com os de um colega que partiu
do mesmo feed}{perceber que cada seleção deixa de fora parte da história, sem
que ninguém tenha mentido}
\aceite
\begin{itemize}
  \item Quando o professor libera o Mural, ele abre na publicação do colega
    que mais se afastou da minha escolha, pelo primeiro nome dele ou pelo nome
    da equipe, e \enquote{Sua publicação} fica ao lado, a um toque. Cada post
    mostra a legenda do veículo sobre a imagem: comparar as seleções é comparar
    o que cada veículo contou. Do mesmo jeito, o colega com
    quem a minha publicação foi pareada vê os três posts que escolhi, a minha
    legenda e o meu primeiro nome (ou o da minha equipe) --- e só isso.
  \item Quando a turma divide computador, o par é de computador: dois
    estudantes do mesmo aparelho nunca são pareados entre si. Sem colega para
    parear --- quem sobrou, quem publicou depois, a aula num computador só ---,
    aparece um colega virtual.
  \item O Mural não tem ficha, nota nem acerto: o contraste vem antes da
    explicação.
  \item Quando o professor revela as fichas técnicas, cada post mostra a fonte
    ouvida e o ângulo, e a ficha compara o equilíbrio editorial das duas
    seleções.
  \item Na explicação, três cartões --- a ilusão da neutralidade, as três lentes
    e quem fala ---, e cada um espera oito segundos antes de deixar avançar.
\end{itemize}
\end{historia}

\begin{historia}{H06.12.4}{Refletir e responder à Amara}{\parcial}
\hu{Estudante do 7º ano}{escrever o que aprendi e responder às perguntas da
Amara}{levar a ideia de recorte para um caso novo}
\aceite
\begin{itemize}
  \item Fora do aplicativo, na tela da escola, \enquote{Reflexão Individual: O
    Papel do Repórter} pede que eu aplique o que vivi a outra cobertura. Escrevo
    de 100 a 500 caracteres, e a tela avisa que só o meu professor lê.
  \item A reflexão não é corrigida: vai para o professor da turma e para mais
    ninguém.
  \item De volta à conversa com a Amara, respondo a duas enquetes: o que eu
    diria a quem achou que os meus Stories contam tudo, e o que pensar da
    seleção do colega. As alternativas mudam de ordem a cada vez, e depois de
    cada uma vejo só \enquote{Resposta enviada} --- nunca se acertei.
  \item A Amara se despede, e a tela da escola mostra \enquote{Missão
    concluída!}.
\end{itemize}
\pendencia{os textos que o roteiro aprovado não tinha e a equipe escreveu ---
entre eles a despedida da Amara, o texto da tela final, a recusa da legenda e
os avisos de conexão --- aguardam a revisão do PO.}
\end{historia}

\begin{historia}{H06.12.5}{Conduzir a aula e ler a turma}{\parcial}
\hu{Professor}{conduzir a turma pelas etapas da aula e ver o que cada
computador publicou}{abrir o debate com o retrato da sala, sem expor ninguém}
\aceite
\begin{itemize}
  \item Um botão só, que diz a próxima etapa --- \enquote{Liberar Mural da
    Turma}, \enquote{Revelar Fichas Técnicas} e \enquote{Liberar Conclusão
    Individual} ---, move a turma inteira, uma etapa por vez, e não volta.
  \item O painel se atualiza sozinho: quantos publicaram os Stories,
    entregaram a reflexão e concluíram. A tela sugere liberar o Mural com cerca
    de 80\% publicados, e quem publica depois ainda ganha um par.
  \item A folha da turma mostra onde cada um está, quando publicou, a seleção
    (quantos posts de cada lente), o par no Mural, a legenda e a reflexão, e o
    perfil editorial médio da sala. O texto da legenda e o da reflexão aparecem
    só para quem dá aula na turma; a gestão vê \enquote{entregou}. Sem nota nem
    acerto.
  \item O bloco \enquote{Habilidades da BNCC} mostra a EF07CO11 pela legenda e
    pela primeira enquete da Amara (vale a melhor das duas) e a EF07CO08 pela
    segunda, e é ali que confirmo e ajusto (\id{E14}). Durante a aula ele se
    atualiza sozinho, também quando a leitura da legenda chega um pouco depois
    da publicação.
  \item A legenda é lida pela IA, que dá o nível dela na EF07CO11 --- a exceção
    declarada à regra de que a IA não avalia (\id{H09.3.1}). Se a IA falhar,
    nada é marcado e vale a enquete; o estudante nunca vê o nível, e eu o
    confiro e ajusto.
  \item \enquote{Zerar a turma} abre uma rodada nova da mesma cobertura e volta
    a aula à exploração. Nada é apagado: a rodada anterior fica no painel de
    habilidades. A janela avisa que, com o mesmo tema, a surpresa do Mural se
    perde, e sugere trocar o tema.
  \item Se a turma divide computadores, monto as equipes na ficha da missão, e
    o painel e a folha contam por computador (\id{E13}).
  \item O guia da aula traz o roteiro, as perguntas do debate e o que fazer com
    menos aparelhos que estudantes.
\end{itemize}
\pendencia{a leitura da legenda pela IA só se confirma com a IA de verdade: no
ambiente de desenvolvimento e no teste de ponta a ponta ela nunca marca, e a
EF07CO11 aparece só pela enquete. Falta conferir, antes da primeira rede, se os
níveis que ela dá são os que o professor daria.}
\end{historia}

\hu{Professor do 7º ano}{um episódio curto em que dois apresentadores
recapitulam a missão com a turma}{fechar a conversa depois da aula, ou
retomá-la na aula seguinte, com a turma ouvindo a mesma coisa}
\aceite
\begin{itemize}
  \item O \enquote{Podcast de recapitulação} fica em \enquote{A aula e os

    recebe ---, e ele decide se e quando toca para a turma.
  \item \enquote{Ler a transcrição} mostra o texto inteiro, bloco a bloco,
    mesmo quando o áudio não existe.
  \item Só as três falas que contam o que cada veículo mostrou mudam com o
    tema. Num tema gerado por IA, elas vêm junto com os posts, e eu as reviso
    antes de aprovar (\id{H06.12.1}); o resto do roteiro não muda.
  \item \enquote{Gerar o áudio do podcast} cria a voz com IA, um trecho por
    vez, e conta no meu limite diário; o que é igual para todas as turmas é
    gerado uma vez só. Se a voz não sair, a tela diz, e a transcrição continua.
\end{itemize}
\pendencia{a voz gerada precisa da revisão do conteúdo antes da primeira rede;
se não passar, o episódio é gravado em estúdio, com o mesmo roteiro.}
\end{historia}

\begin{historia}{H06.12.7}{Aprovar as imagens antes de abrir a turma}{\parcial}
\hu{Professor}{ver cada imagem junto do seu post e trocar a que não serve antes
de a turma entrar}{que nenhum estudante veja uma imagem que eu não vi}
\aceite
\begin{itemize}
  \item Aprovados os textos (\id{H06.12.1}), as imagens são criadas sozinhas,
    na mesma tela, cada uma no card do seu post, ao lado da legenda --- vejo a
    composição que o estudante vai ver. Os posts de números já vêm com gráfico.
  \item \enquote{Gerar outra imagem} troca a imagem de um post só. Cada troca
    conta no meu limite diário, e só vale antes da aprovação.
  \item O último passo da criação, \enquote{Aprovar Cobertura e Abrir Turma}
    (ou \enquote{Aprovar cobertura}, se eu não for pôr no ar agora), só acende
    depois das duas fases. Ele aprova as imagens e, com \enquote{Pôr esta
    atividade no ar agora} marcado, abre a turma (\id{H06.13.3}).
  \item Se a imagem de um post não sair, ele segue com a descrição da cena, que
    eu vejo no mesmo card antes de aprovar; a missão roda igual.
  \item Se eu sair antes de aprovar, a atividade fica \enquote{Em preparo}, sem
    \enquote{Usar esta atividade agora}; a página dela mostra os mesmos cards,
    com \enquote{Aprovar imagens}. Aprovadas, as imagens não mudam mais: a
    turma pode estar olhando para elas.
\end{itemize}
\pendencia{as imagens geradas precisam da revisão do conteúdo antes da
primeira rede.}
\end{historia}

\feature{F06.13}{As atividades da missão}{\pronto}

A missão é o roteiro; a \textbf{atividade} é a missão que a turma faz de fato,
com o conteúdo que o professor escolheu. Vale para o Unoovista, A Seleção, o
Jornal da Escola, Quem mandou isso? e O Júri Estadual de Projetos: a ficha de cada uma tem a mesma lista, no
molde da que o Unoovista já tinha. A plataforma não cria atividade nenhuma
sozinha.

\begin{historia}{H06.13.1}{Ver as atividades da turma}{\pronto}
\hu{Professor}{ver, na ficha da missão, as atividades que criei para a turma}{saber
o que a turma está fazendo e o que está pronto para a próxima aula}
\aceite
\begin{itemize}
  \item \enquote{Suas atividades com a turma\ldots} lista cada atividade com o
    tema, de onde veio o conteúdo, quantas rodadas já rodou e quando foi criada,
    e o estado: \enquote{Em andamento}, \enquote{Pronta} ou, no Unoovista, na
    A Seleção e no Jornal da Escola, \enquote{Em preparo}. A em andamento vem primeiro.
  \item Aberta sem turma, a ficha pede para escolher uma: a atividade é sempre
    de uma turma.
  \item Sem nenhuma em andamento, a ficha avisa que a turma vê a missão fechada.
\end{itemize}
\end{historia}

\begin{historia}{H06.13.2}{Criar a atividade em quatro passos}{\pronto}
\hu{Professor}{criar uma atividade escolhendo a turma, o conteúdo e como a sala
vai se sentar}{montar a aula inteira num lugar só}
\aceite
\begin{itemize}
  \item \enquote{Nova atividade} abre os passos \enquote{Turma},
    \enquote{Conteúdo}, \enquote{Equipes} e \enquote{Revisar e criar}; a turma
    já vem escolhida quando eu chego pela sala.
  \item O conteúdo é o de cada missão: gerado com IA sobre o meu tema, ou a
    partir do conteúdo pronto da Unooria, que eu leio e edito antes. No Júri
    Estadual, a mostra pronta fica a um toque (\id{H06.14.1}).
  \item As equipes se escolhem ali mesmo (\id{H13.1.8}).
  \item Nada é gravado antes de \enquote{Criar atividade} --- exceto na A
    Seleção e no Jornal da Escola, em que os textos são guardados ao serem
    aprovados, para as imagens serem criadas a partir deles (\id{H06.12.7},
    \id{H06.10.6}). Se as equipes não puderem ser salvas, a atividade é criada
    mesmo assim, e a tela diz o que faltou.
  \item O Unoovista mantém a criação dele, porque os posts levam alguns
    minutos para ficar prontos; as equipes entram na página da atividade, antes
    de \enquote{Ativar}.
\end{itemize}
\end{historia}

\begin{historia}{H06.13.3}{Escolher qual atividade a turma faz}{\pronto}
\hu{Professor}{escolher qual das minhas atividades a turma faz agora}{preparar a
próxima aula sem tirar do ar a de hoje}
\aceite
\begin{itemize}
  \item Criar não põe a atividade no ar: ela fica \enquote{Pronta}. No último
    passo da criação posso marcar \enquote{Pôr esta atividade no ar agora}, que
    já vem marcado quando a turma não tem nada em andamento.
  \item \enquote{Usar esta atividade agora} põe a escolhida em andamento. Se a
    turma estiver fazendo outra, a tela pergunta antes: quem está jogando passa
    para a nova, e a outra volta a ficar pronta, com tudo guardado.
  \item Pôr no ar --- pelo último passo da criação ou pelo botão --- também
    abre a missão para a turma na sala de aula (\id{H04.2.1}), mesmo que eu a
    tenha encerrado lá. Antes, a atividade ficava no ar e a turma via a missão
    fechada. Se só a abertura na sala falhar, a tela avisa e diz onde
    terminar.
  \item Uma atividade em andamento por turma. No Unoovista, duas turmas do
    mesmo professor podem rodar ao mesmo tempo.
  \item Voltar a uma atividade que já rodou começa uma rodada nova; as
    anteriores continuam no painel de habilidades.
\end{itemize}
\end{historia}

\begin{historia}{H06.13.4}{A página da atividade}{\pronto}
\hu{Professor}{abrir uma atividade e ter nela tudo o que faço com ela}{preparar e
conduzir a aula sem procurar em outra tela}
\aceite
\begin{itemize}
  \item A página mostra o tema e o estado, as equipes da atividade, as imagens
    (quando a missão tem) e o painel de habilidades.
  \item O painel da aula, com os botões que avançam a turma e o
    \enquote{Zerar}, só aparece na atividade em andamento; na pronta, a página
    oferece \enquote{Usar esta atividade agora}. Na A Seleção, a atividade em
    preparo mostra, no lugar dele, os cards dos posts com \enquote{Aprovar
    imagens} (\id{H06.12.7}); no Jornal da Escola, o aviso \enquote{Aprove as
    imagens antes de abrir a turma} e as imagens das notícias antes das
    equipes (\id{H06.10.6}).
\end{itemize}
\end{historia}

\feature{F06.6}{Unoovista --- O conteúdo que não chegou}{\pronto}

\textbf{8º ano EF, Unidade 1} \textperiodcentered{} imita uma rede social de
fotos, em moldura de celular \textperiodcentered{} uma aula
\textperiodcentered{} IA no preparo e nas reações ao vivo.

\textbf{Proposta.} Mostrar como um feed decide o que cada pessoa vê e o que
isso faz com uma opinião em formação (a bolha). Desde outubro de 2026 o
aplicativo imitado se chama \textbf{Unoovista} e tem a identidade visual da
Unooria (roxo, cartões, a barra de navegação flutuando); até ali era o
Instanooria, que copiava a cara de um aplicativo real. O que o estudante faz
e o que o feed mede não mudaram --- e A Seleção mudou-se para o Unoovista no
mesmo dia. Foi a primeira missão com
\enquote{atividade} configurável pelo professor; hoje Quem mandou isso?, A
Seleção, o Jornal da Escola e O Júri Estadual de Projetos também têm a sua
(\id{E14}). O Júri Estadual, do 6º ano, mostra o mesmo fenômeno por outro eixo
--- as áreas de uma feira, e não os dois lados de um tema --- e ensina a regra
do feed com todas as letras (\id{F06.14}).

\begin{historia}{H06.6.1}{Criar a atividade a partir da minha aula}{\pronto}
\hu{Professor}{criar uma atividade sobre o tema que estou trabalhando}{que o
feed discuta um assunto da minha disciplina}
\aceite
\begin{itemize}
  \item \enquote{Nova atividade} pede Título, Disciplina, \enquote{Tema /
    recorte}, turma e \enquote{O que já foi trabalhado em aula (opcional)}.
  \item A pergunta-chave não é digitada: a IA propõe candidatas, cada uma com
    uma nota de equilíbrio entre os dois lados.
  \item Escolher uma pergunta desequilibrada exige que eu justifique por
    escrito.
  \item Valido os dois lados (Tese, \enquote{Argumentos (um por linha)},
    \enquote{Tom típico}).
\end{itemize}
\end{historia}

\begin{historia}{H06.6.2}{Revisar o conteúdo antes de ativar}{\pronto}
\hu{Professor}{revisar os posts gerados antes da aula}{garantir que nada
inadequado chegue à turma}
\aceite
\begin{itemize}
  \item A geração acontece em fila (Perfis, Posts, Imagens, Revisão) e a
    atividade passa por Rascunho, Aguardando validação, Gerando conteúdo, Em
    revisão, Ativa e Encerrada.
  \item Filtro os posts por lado, perfil, intensidade, categoria e imagem;
    posso regerar texto ou imagem, ou excluir.
  \item Imagem que falhou recebe uma arte substituta, e a atividade não trava.
  \item Com conteúdo de menos, a plataforma recusa ativar e lista o que falta.
  \item Nenhuma criança entra antes de \enquote{Ativar}.
\end{itemize}
\end{historia}

\begin{historia}{H06.6.3}{Decidir de que lado cada estudante começa}{\pronto}

\end{historia}

\begin{historia}{H06.6.11}{O Raio-x do feed, antes do debate}{\pronto}

\feature{F06.9}{Vila do Lume --- A Noite Longa}{\pronto}

\textbf{Para qualquer ano} (6 a 13 anos) \textperiodcentered{} jogo 3D próprio,
em primeira pessoa \textperiodcentered{} uma aula.

\textbf{Proposta.} Na camada visível, reconhecer um golpe que não tem os sinais
que se costuma ensinar a procurar. Na camada invisível, um retrato de valores
feito a partir de escolhas que custam algo --- e que define de que lado cada
estudante começa no Unoovista.

\begin{historia}{H06.9.1}{Acender as fogueiras da vila}{\pronto}
\hu{Estudante}{explorar a vila e acender as oito fogueiras}{salvar a vila antes

\aceite
\begin{itemize}
  \item Uma instrução só: \enquote{A vila apagou. Acenda as oito fogueiras
    antes que a noite acabe.}
  \item Pelo caminho há gestos opcionais (ajudar um aldeão, acender um
    lampião, gravar o nome na pedra, tocar o sino), que não valem ponto.
  \item O nome da criança nunca é gravado.
\end{itemize}
\end{historia}

\begin{historia}{H06.9.2}{Receber mensagens no meio do jogo}{\pronto}
\hu{Estudante}{receber mensagens enquanto jogo}{descobrir, no debate, em quais
eu confiei}
\aceite
\begin{itemize}
  \item Chegam 5 mensagens sem pausar o jogo: verdadeiras e um golpe principal
    num de três níveis (grosseiro, autoridade com urgência, colega pedindo um
    passo pequeno).
  \item O nível é distribuído de forma equilibrada na turma; do 1º ao 5º ano o
    nível intermediário não é usado.
  \item O jogo não tem tela de resultado: a revelação é do professor.
\end{itemize}
\end{historia}

\begin{historia}{H06.9.3}{Ler a turma sem rotular ninguém}{\pronto}
\hu{Professor}{entender o que aconteceu com as mensagens na
turma}{conduzir o debate \enquote{Em quem dava pra confiar?}}
\aceite
\begin{itemize}
  \item Na revelação, projeto as mensagens das três versões lado a lado.
  \item Resultados: \enquote{As três bolhas nesta turma}, \enquote{O que
    aconteceu com as mensagens} e \enquote{Estudante por estudante} --- sem
    nenhum valor pessoal na lista.
  \item O relatório de uma criança abre pelo nível do golpe que ela recebeu e
    mostra \enquote{Os dez valores}; só há número quando há dois ou mais
    sinais, e \enquote{não teve a oportunidade} é diferente de \enquote{teve e
    não foi}.
  \item Partida abandonada não gera retrato, e a tela explica por quê.
  \item \enquote{Zerar a turma} (\enquote{Isso não tem volta.}) também
    redistribui as versões.
  \item A gestão vê o resumo agregado; a linha de uma criança, não.
\end{itemize}
\pendencia{o retrato de valores é uma hipótese de desenho, não um instrumento
validado. As reações dentro do jogo ainda não contam no limite diário de IA.}
\end{historia}

\feature{F06.10}{As missões que faltam}{\afazer}

A trilha tem 48 unidades (12 anos $\times$ 4). Estão publicadas 9, mais a Vila
do Lume. O 2º, o 3º e o 9º ano do Fundamental e a 1ª e a 3ª série do Médio
ainda não têm nenhuma. Em 25 de setembro de 2026 três anos ficaram sem a sua
missão por alguns dias: o 4º, até o Jornal da Escola ocupar o lugar do UnooTube
arquivado (\id{H06.10.4}); o 5º, até a versão nova de Quem mandou isso? voltar

(\id{F06.12}). As unidades em produção aparecem para o professor como

\begin{historia}{H06.10.1}{Cada missão nova nasce completa}{\afazer}

  \item A correção é da plataforma; os resultados não têm ranking nem coluna de
    acerto.
  \item O texto livre da criança é lido só pelo professor da turma.
  \item Se usa IA: nada gerado chega à turma sem aprovação, há material pronto
    se a IA falhar, e o uso conta no limite diário.
  \item Toda a interface em português, exceto os termos do aplicativo imitado.
  \item Tem teste automatizado que percorre a missão como o estudante e como o
    professor.
\end{itemize}
\end{historia}

\begin{historia}{H06.10.4}{Jornal da Escola (4º ano, Unidade 1)}{\parcial}
\hu{Estudante do 4º ano}{ser editor do jornal da escola: escolher três notícias
entre oito, publicar, ver o que aconteceu com os leitores e publicar de novo}{aprender
a conferir fonte, data e autoria antes de informar outras pessoas}
\aceite
\begin{itemize}
  \item Ocupa o lugar do UnooTube, que foi arquivado em 25 de setembro de 2026.
  \item Entro na redação do jornal da escola, que é da própria Unooria e não
    imita rede social nenhuma. O Bento, que cuida do jornal, me apresenta o
    desafio: três vagas na primeira página.
  \item Na mesa há oito notícias de oito meios diferentes (jornal, rádio,
    telejornal, blog, portal, rede social, mensagem e mural). Ao abrir uma, vejo
    a matéria como ela chegou: o post aparece como post do Unoovista, a rede
    social de fotos, e a mensagem encaminhada como mensagem do Unoochat, o
    aplicativo de conversa --- sem curtidas e sem horário. Desde outubro de 2026
    são as redes com a identidade visual da Unooria, as mesmas das outras
    missões; até ali eram o Instanooria e o clone do WhatsApp.
  \item Em cada notícia posso abrir três pistas: de onde veio, quando aconteceu
    e quem assina. Ninguém me obriga a abrir. Logo depois de ler uma pista, digo
    se confio nela ou não (\id{H06.10.8}).
  \item Escolho três notícias. Quando a terceira entra, a mesa me leva sozinha
    para a aba \enquote{Primeira Página}, onde está o botão de publicar --- não
    fico parado nas pautas sem saber o próximo passo.
  \item Publico a primeira edição. Depois de publicar, não dá para trocar; só
    então a missão pergunta \enquote{Como as três matérias foram
    escolhidas?} --- \enquote{porque chamou atenção} ou \enquote{porque conferi
    as pistas}. A pergunta não fala em dupla: quantos dividem o computador é o
    professor quem decide.
  \item O professor libera o resultado: vejo o que os leitores da escola
    disseram das três notícias publicadas e quantas estavam verificadas. O
    resultado fala da edição, nunca de mim, e ninguém vê a edição dos colegas.
  \item Depois da conversa com a turma --- que pode ficar para a aula seguinte
    ---, o professor libera a revisão. Monto a Chave do Editor com três blocos
    entre seis, de uma vez só (\id{H06.10.8}).
  \item Com a Chave, a mesa volta já aberta na \enquote{Primeira Página}, com
    um aviso em cada vaga da primeira edição e o recado \enquote{A Chave do
    Editor marcou notícias duvidosas. Troque os furos por matérias
    verificadas!}; o \enquote{Trocar Matéria} está ali mesmo. Publico a segunda
    edição e vejo o antes e o depois, lado a lado.
  \item No fim, testo a minha Chave em duas matérias novas e recebo uma
    mensagem urgente que posso repassar ou não (\id{H06.10.8}). A missão não me
    diz se acertei.
  \item O professor escolhe o conteúdo antes da aula: o banco padrão da Unooria
    (\enquote{Feira de Ciências e Notícias da Cidade}) ou oito notícias geradas
    por IA sobre o tema dele, que ele revisa, edita notícia a notícia e aprova.
    Se a IA dos textos falhar ou o limite do dia acabar, o banco padrão fica a
    um toque. Junto com os textos, ele revisa as três falas do podcast de
    recapitulação que contam a mesa (\id{H06.10.7}). Depois dos textos, ele vê
    e aprova cada imagem (\id{H06.10.6}): imagem que ele não viu não chega à
    mesa das crianças.
  \item A missão aceita equipes num computador dividido (\id{E13}) e registra o
    que cada estudante demonstrou de três habilidades da BNCC (\id{E14}).
\end{itemize}
\pendencia{as imagens das notícias do banco padrão ainda não foram geradas nem
revisadas; o critério do livro do 4º ano --- fonte, data e autoria --- é
provisório e será revisado pelo dono do produto; e os Termos de Uso ainda não
citam a IA do Jornal. O teste que percorre a missão como estudante e professora
já roda verde (27 de setembro de 2026).}
\end{historia}

\begin{historia}{H06.10.8}{Jornal da Escola: só pelo que a criança faz, sem pergunta de múltipla escolha}{\parcial}
\hu{Professor do 4º ano}{saber o que cada estudante demonstrou das três
habilidades pelo que ele FEZ na missão, e não por uma pergunta de marcar}{confiar
no nível que leio no painel de habilidades}
\aceite
\begin{itemize}
  \item \textbf{Na mesa}, logo depois de abrir uma pista, o estudante marca
    \enquote{Confio} ou \enquote{Não confio}, sempre com o texto junto do
    símbolo. Marcar é opcional, e a primeira marca vale. É assim que se vê se
    ele divide o problema em partes e resolve cada uma: perceber a pista que
    falha, e não pôr na capa a notícia em que ele mesmo não confiou.
  \item \textbf{A Chave do Editor é montada, de uma vez}: três blocos entre
    seis (fonte, data e autoria contra três armadilhas: a foto bonita, a
    urgência e o \enquote{muita gente compartilhou}); em cada um, quando a pista
    passa e quando falha, com um exemplo curto; o que fazer quando ela falha
    (\enquote{Não publico} ou \enquote{Olho a próxima pista}); e se os passos
    valem para todas as notícias ou só para a primeira. Um lembrete pede que a
    dupla combine antes de tocar, e o \enquote{Pronto} só acende com tudo
    escolhido. Ninguém diz se a Chave está certa.
  \item \textbf{Depois da segunda edição}, ele testa a Chave dele em duas
    matérias novas e curtas: abre as pistas na ordem que quiser, marca se
    confia em cada uma e decide publicar ou não. Conta se ele seguiu a própria
    regra. No fim, todos recebem a mesma frase com a regra de referência --- a
    Chave montada errada não sai da aula reforçada.
  \item \textbf{A mensagem urgente} (\enquote{a água de todas as escolas do
    bairro vai ser cortada amanhã}) chega com \enquote{Repassar para o grupo}
    à mão desde o início e as pistas para abrir, numa tela sem destaque nem fala
    do Bento. Conta se ele confere a mensagem antes de decidir e não a repassa.
    A síntese diz que a turma vai conversar sobre a mensagem com a professora.
  \item As duas perguntas de múltipla escolha do desafio final saíram. A
    pergunta de como as notícias foram escolhidas continua, e não vale nível.
  \item Na criação, o professor revisa também as duas matérias da simulação
    (sem imagem), junto das oito. No painel da aula ele vê só números da turma:
    as armadilhas mais escolhidas na Chave, quantos escolheram \enquote{Olho a
    próxima pista} e \enquote{Só na primeira notícia}, e quantos repassaram a
    mensagem sem abrir nenhuma pista. O nível de cada estudante fica no painel
    de habilidades, só para ele.
  \item A missão cabe em uma ou duas aulas, com uma pausa depois do resultado.
\end{itemize}
\pendencia{os textos novos (a Chave, a simulação, a mensagem) esperam a revisão do
dono do produto; o teste de ponta a ponta precisa percorrer a jornada nova com a
IA de verdade na cópia principal; e o tempo real da aula só se sabe numa turma
piloto.}
\end{historia}

\begin{historia}{H06.10.6}{Jornal da Escola: aprovar cada imagem antes de abrir a turma}{\parcial}
\hu{Professor do 4º ano}{ver cada imagem das notícias junto da manchete, trocar a
que não serve e aprovar uma a uma antes de a turma entrar}{que nenhum estudante
veja uma imagem gerada por IA que eu não vi}
\aceite
\begin{itemize}
  \item A criação tem duas fases, como na A Seleção (\id{H06.12.7}). Na
    primeira, os textos: \enquote{Aprovar textos e gerar as imagens} guarda as
    oito notícias. Na segunda, as imagens: elas são criadas sozinhas, na mesma
    tela, cada uma ao lado da manchete da notícia, com \enquote{Aguardando seu
    OK} até eu aprovar.
  \item \enquote{Gerar outra imagem} troca a imagem de uma notícia só, e ela
    volta a esperar o meu OK. Cada troca conta no meu limite diário.
  \item \enquote{Aprovar} aprova as imagens que estão na tela. O
    \enquote{Continuar} da criação só acende com todas aprovadas.
  \item \textbf{A imagem não é opcional.} Sem todas as imagens geradas e
    aprovadas, a atividade não vai ao ar: na lista ela aparece \enquote{Em
    preparo}, sem \enquote{Usar esta atividade agora}, e a página dela mostra
    \enquote{Aprove as imagens antes de abrir a turma} e as imagens antes das
    equipes. Se a IA de imagens estiver fora do ar ou o limite do dia acabar, a
    atividade nova espera.
  \item Com a atividade já no ar, também posso trocar uma imagem: só aquela
    notícia fica sem imagem para a turma (com o desenho do meio de onde veio)
    até eu aprovar a nova; as outras continuam, e a aula não para.
  \item Uma atividade nova aproveita as imagens já criadas para a mesma
    descrição, mas nunca a aprovação: vejo e aprovo as imagens de cada
    atividade.
  \item As atividades criadas antes desta regra continuam com as imagens que
    tinham, já aprovadas.
\end{itemize}
\pendencia{o teste de ponta a ponta que percorre as duas fases com a IA de
verdade foi escrito com a mudança e precisa rodar verde na cópia principal
para a história passar a pronta.}
\end{historia}

\begin{historia}{H06.10.7}{Jornal da Escola: o podcast de recapitulação}{\parcial}
\hu{Professor do 4º ano}{um episódio curto em que dois locutores da rádio da
escola recapitulam a missão com a turma, contando a mesa que ela viu}{fechar a
conversa depois da aula, ou retomá-la na aula seguinte, com a turma ouvindo a
mesma coisa}
\aceite
\begin{itemize}
  \item O episódio tem cerca de 3 minutos e 30 segundos, com Lucas e Clara, e
    segue o roteiro aprovado para a missão. É do professor, como o da A
    Seleção (\id{H06.12.6}): nenhum estudante o recebe, e ele decide se e
    quando toca para a turma.
  \item Só três falas mudam com o tema, porque contam o que a turma viu: a
    notícia antiga que mais chamava a atenção, a reação dos leitores a ela e o
    aviso sem assinatura. O resto do roteiro é igual para todas as turmas.
  \item \textbf{Reviso as três falas ao criar a atividade}, abaixo da mesa de
    notícias: quando as oito notícias chegam à tela, as falas chegam junto,
    escritas pela IA a partir delas, e eu posso editar cada uma ou pedir que
    sejam escritas de novo. Com o banco padrão, sem mudança, as falas são as
    do roteiro aprovado, sem IA. \enquote{Aprovar textos} espera as falas, e
    elas são aprovadas junto com as notícias.
  \item Se o limite do dia acabar ou a IA estiver fora do ar, a tela traz as
    falas do banco padrão e avisa que elas contam outra mesa, para eu
    reescrever.
  \item Se eu mudar as notícias depois de as falas serem escritas, a tela avisa
    que elas podem não contar mais o que a turma vai ver.
  \item Na página da atividade, em \enquote{A aula e os resultados}, fica o
    podcast: a transcrição inteira, bloco a bloco, mesmo sem áudio, e
    \enquote{Gerar o áudio do podcast}, que cria a voz com IA um trecho por
    vez e conta no meu limite diário. O que é igual para todas as turmas é
    gerado uma vez só.
  \item Uma atividade criada com IA antes de o podcast existir não tem falas
    aprovadas: a página dela diz que é preciso criar a atividade de novo para
    ter o podcast.
  \item O guia da aula traz o passo \enquote{Gere o áudio do podcast, se for
    usá-lo}.
\end{itemize}
\pendencia{a voz gerada precisa da revisão do conteúdo antes da primeira rede;
se não passar, o episódio é gravado em estúdio, com o mesmo roteiro.}
\end{historia}
