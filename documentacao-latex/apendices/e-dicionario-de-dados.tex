\begin{decisao}
\cd{UNIQUE (escola\_id, estudante\_id, emissor\_contrato, emitido\_em)}: a escola
recebe o arquivo por e-mail e alguém importa duas vezes. Duas cópias do mesmo
fato pareceriam duas medidas. Reimportar reconfere a assinatura e atualiza só o
veredito --- a rede pode ter voltado, e \enquote{não consegui falar com o
emissor} deixar de valer não é fato novo sobre a criança.
\end{decisao}

\textbf{O medidor de minutos da Ajuda com avatar} (\adrr{ADR-132}): uma linha

\section{O que o Prisma não modela}

\begin{atencao}
Ler apenas \arq{schema.prisma} dá uma visão \textbf{incompleta} do banco. Vivem
\emph{só} na migração:

\begin{itemize}
  \item políticas de \textbf{RLS} (\cd{ENABLE}, \cd{FORCE}, \cd{CREATE POLICY});
  \item \textbf{\emph{CHECK} constraints} (papel \emph{versus} \cd{escola\_id},
    cardinalidade dos salvos, vocabulário de \cd{permissoes}, coerência de
    conclusão, XOR de ancoragem de comentário);
  \item \textbf{partições} e o \emph{schema} \cd{particoes};
  \item \textbf{\cd{GRANT}} para a \emph{role} de aplicação.
\end{itemize}

Os campos correspondentes são \cd{TEXT}/\cd{SMALLINT} no \emph{schema}, e a
validação de aplicação é feita com Zod.
\end{atencao}
