\section{Evolução planejada}

\subsection{Antes de novembro de 2026 (bloqueante)}

\begin{enumerate}
  \item \textbf{Teste de carga com k6}: \num{4000} \emph{logins} e aberturas de
    experiência em dez minutos, contra a \emph{stack} \cd{dev}. Verificar
    saturação de conexões, latência do \rota{/runs/:id/feed} sob \emph{lock}, e o
    caminho de ingestão de telemetria.
  \item \textbf{Observabilidade, fase 0} (Capítulo~\ref{cap:observabilidade}):
    canais de notificação, \emph{uptime checks} do JWKS e de \rota{/readyz},
    SLO de disponibilidade com \emph{burn rate}, falha do Cloud Run Job de
    migração, saturação do Cloud SQL, \emph{dead letter} da telemetria e
    \emph{smoke} pós-\emph{deploy} --- tudo em Terraform e no
    \arq{deploy.yml}, sem tocar em código de aplicação. A fase 1 (log
    estruturado com \emph{redaction} e \cd{ExceptionFilter}) entra logo
    depois, porque é o que torna o alerta diagnosticável.
  \item \textbf{Decisão de retenção}, com a rotina que a cumpre. Se a resposta for
    \enquote{sem prazo}, ela precisa estar no contrato --- não só neste documento.
  \item \textbf{Revisão de segurança externa} e teste de intrusão.
  \item \textbf{DPA por rede} contemplando a localidade de processamento de IA.
  \item \textbf{WSO2 como emissor}, com o \cd{iss} \emph{tenant-qualified} copiado
    literalmente do \emph{discovery}.
\end{enumerate}
